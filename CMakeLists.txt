cmake_minimum_required(VERSION 3.21)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>" CACHE STRING "")
project(TFCam VERSION 0.17.0 LANGUAGES CXX)

if(DEFINED ENV{SKYRIM_MODS_FOLDER} AND IS_DIRECTORY "$ENV{SKYRIM_MODS_FOLDER}")
    set(OUTPUT_FOLDER "$ENV{SKYRIM_MODS_FOLDER}/FreeCam--Claude")
endif()

# 0.8.0: CommonLibSSE-NG from alandtse (github.com/alandtse/CommonLibSSE-NG, formerly CommonLibVR, branch `ng`),
# built from source out of the pinned submodule extern/CommonLibSSE-NG (tag v9.1.0). This is the lineage that reads
# the Address Library v5 database and sets the AddressLibraryV5 flag, so 1.7.x SKSE loads the DLL. See BUILD.md.
set(CLIB_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/extern/CommonLibSSE-NG")
if(NOT EXISTS "${CLIB_SOURCE_DIR}/cmake/CommonLibSSE.cmake")
    message(FATAL_ERROR "extern/CommonLibSSE-NG is empty. Run: git submodule update --init --recursive --depth 1")
endif()
if((NOT TFCAM_RUNTIME OR TFCAM_RUNTIME STREQUAL "ALL") AND
   NOT EXISTS "${CLIB_SOURCE_DIR}/extern/openvr/lib/win64/openvr_api.lib")
    message(FATAL_ERROR "extern/CommonLibSSE-NG/extern/openvr is empty (VR is enabled). Run: git submodule update --init --recursive --depth 1")
endif()

# Every CACHE FORCE must precede add_subdirectory: option() inside NG cannot override an existing cache entry.
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)                 # NG defaults ON -> find_package(Catch2 REQUIRED)
set(SKSE_SUPPORT_XBYAK OFF CACHE BOOL "" FORCE)          # TFCam uses no trampoline
set(SKSE_SUPPORT_PATCH_SAFETY OFF CACHE BOOL "" FORCE)   # write_branch diagnostics; TFCam has no write_branch, and ON
                                                         # FetchContent-downloads minhook at configure time
# TFCAM_RUNTIME picks which game runtimes the DLL is built for (0.14.1):
#   ALL (default) = SE + AE + VR in one DLL, the same as the 0.7.x - 0.14.0 builds (runtime dispatch)
#   SE = Skyrim SE 1.5.97 only, AE = Skyrim AE 1.6.x and 1.7.x only
# A VR-only build is not offered: FreeCamController.cpp pins the SE / AE field offsets of RE::FreeCameraState and the AE 1.7
# virtual-slot shift, and both static_asserts fail against the VR layout. VR stays in the ALL build, as before.
# The release installer ships ALL, SE and AE; see BUILD.md "Runtime variants". Configure each in its own build folder.
set(TFCAM_RUNTIME "ALL" CACHE STRING "Game runtimes to build for: ALL, SE or AE")
set_property(CACHE TFCAM_RUNTIME PROPERTY STRINGS ALL SE AE)
if(NOT TFCAM_RUNTIME MATCHES "^(ALL|SE|AE)$")
    message(FATAL_ERROR "TFCAM_RUNTIME must be ALL, SE or AE (got '${TFCAM_RUNTIME}')")
endif()
set(_rt_se OFF)
set(_rt_ae OFF)
set(_rt_vr OFF)
if(TFCAM_RUNTIME STREQUAL "ALL" OR TFCAM_RUNTIME STREQUAL "SE")
    set(_rt_se ON)
endif()
if(TFCAM_RUNTIME STREQUAL "ALL" OR TFCAM_RUNTIME STREQUAL "AE")
    set(_rt_ae ON)  # AE = 1.6.x AND 1.7.x (NG classifies minor >= 6 as AE)
endif()
if(TFCAM_RUNTIME STREQUAL "ALL")
    set(_rt_vr ON)
endif()
set(ENABLE_SKYRIM_SE ${_rt_se} CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_AE ${_rt_ae} CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_VR ${_rt_vr} CACHE BOOL "" FORCE)
message(STATUS "TFCam runtimes: ${TFCAM_RUNTIME} (SE ${_rt_se}, AE ${_rt_ae}, VR ${_rt_vr})")
set(REX_OPTION_INI OFF CACHE BOOL "" FORCE)
set(REX_OPTION_JSON OFF CACHE BOOL "" FORCE)
set(REX_OPTION_TOML OFF CACHE BOOL "" FORCE)

add_subdirectory("${CLIB_SOURCE_DIR}" CommonLibSSE-NG)
# project() inside the subdirectory keeps CommonLibSSE_VERSION in that scope; read the pinned version from its manifest.
file(READ "${CLIB_SOURCE_DIR}/vcpkg.json" _clib_manifest)
string(JSON TFCAM_CLIB_VERSION GET "${_clib_manifest}" "version-semver")
message(STATUS "CommonLibSSE-NG ${TFCAM_CLIB_VERSION} from ${CLIB_SOURCE_DIR}")
# NG's source build does not include its plugin helper (only the prebuilt path does).
include("${CLIB_SOURCE_DIR}/cmake/CommonLibSSE.cmake")

# Declaration generated by add_commonlibsse_plugin (SKSEPlugin_Version + SKSEPlugin_Query):
#   - Address Library independence (versionIndependence = AddressLibraryPostAE), no fixed runtime list
#   - StructCompatibility::Independent = AddressLibraryV5 | NoStructUse in versionIndependenceEx
# MINIMUM_SKSE_VERSION must be a full x.y.z triplet: a bare "0" (the helper's default) generates
# REL::Version{ 0, , , 0 } and fails to compile.
add_commonlibsse_plugin(${PROJECT_NAME} MINIMUM_SKSE_VERSION 0.0.0 SOURCES
    src/plugin.cpp
    src/FreeCamController.cpp
    src/FreeCamMenu.cpp
    src/FreezeTime.cpp
    src/CameraLight.cpp
    src/HUDHider.cpp
    src/FcfwBridge.cpp
    src/SlccBridge.cpp
    src/SceneTracker.cpp
    src/CellEntryCam.cpp
    src/TFCamAPI.cpp
    src/SceneCam.cpp
    src/ConvoCam.cpp
    src/SceneShots.cpp
    src/AmbientCam.cpp
    src/CamCollision.cpp
    src/CamFocus.cpp
    src/CamBlur.cpp
    src/CamPath.cpp
)
target_compile_features(${PROJECT_NAME} PRIVATE cxx_std_23)
target_precompile_headers(${PROJECT_NAME} PRIVATE PCH.h)

# Logged at load: "built on CommonLibSSE-NG <version>".
target_compile_definitions(${PROJECT_NAME} PRIVATE
    TFCAM_VERSION="${PROJECT_VERSION}"
    TFCAM_COMMONLIB_VERSION="${TFCAM_CLIB_VERSION}"
    TFCAM_RUNTIME_NAME="${TFCAM_RUNTIME}"
)

target_include_directories(${PROJECT_NAME} PRIVATE
    ${CMAKE_SOURCE_DIR}/src
    ${CMAKE_SOURCE_DIR}/include
)

if(DEFINED OUTPUT_FOLDER)
    # Copies TFCam.dll (and the PDB in Debug) only. TFCam.ini is the user's settings file: TFCam creates it on first
    # run and no build, deploy or release zip ever ships or copies it (0.7.9).
    set(DLL_FOLDER "${OUTPUT_FOLDER}/SKSE/Plugins")

    message(STATUS "SKSE plugin output: ${DLL_FOLDER}")

    add_custom_command(
        TARGET "${PROJECT_NAME}" POST_BUILD
        COMMAND "${CMAKE_COMMAND}" -E make_directory "${DLL_FOLDER}"
        COMMAND "${CMAKE_COMMAND}" -E copy_if_different "$<TARGET_FILE:${PROJECT_NAME}>" "${DLL_FOLDER}/$<TARGET_FILE_NAME:${PROJECT_NAME}>"
        VERBATIM
    )

    if(CMAKE_BUILD_TYPE STREQUAL "Debug")
        add_custom_command(
            TARGET "${PROJECT_NAME}" POST_BUILD
            COMMAND "${CMAKE_COMMAND}" -E copy_if_different "$<TARGET_PDB_FILE:${PROJECT_NAME}>" "${DLL_FOLDER}/$<TARGET_PDB_FILE_NAME:${PROJECT_NAME}>"
            VERBATIM
        )
    endif()
endif()
